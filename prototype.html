// npm install @arkreen/a4a-sdk
import { A4AClient } from '@arkreen/a4a-sdk';

const client = new A4AClient({
  apiKey: 'a4a_sk_live_...',
  agentWallet: '0x3aF2...9b1c',
});

// trigger a run
const run = await client.triggerRun({ strategy: 'dca' });
// { id: '#run-0492', status: 'CREATED' }

// poll or listen
client.on('COMPLETED', (e) => {
  console.log(`✓ offset ${e.offsetKwh} kWh`);
});
